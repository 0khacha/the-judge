import time
from typing import Dict, Optional

class JWTTokenManager:
    def __init__(self, secret: str, ttl_seconds: float = 5.0):
        self.secret = secret
        self.ttl = ttl_seconds
        self.tokens: Dict[str, float] = {}
        self.revoked = set()

    def issue_token(self, user_id: str) -> str:
        token = f"token_{user_id}_{len(self.tokens)}_{time.time()}"
        self.tokens[token] = time.time()
        return token

    def validate_token(self, token: str) -> bool:
        if token in self.revoked or token not in self.tokens:
            return False
        issue_time = self.tokens[token]
        if time.time() - issue_time >= self.ttl:
            return False
        return True

    def revoke_token(self, token: str) -> None:
        self.revoked.add(token)
