class TransactionDB:
    def __init__(self):
        self.data = {}
        self.checkpoint = None

    def begin(self):
        self.checkpoint = dict(self.data)

    def put(self, key: str, val: int):
        self.data[key] = val

    def get(self, key: str):
        return self.data.get(key)

    def rollback(self):
        if self.checkpoint is not None:
            self.data = dict(self.checkpoint)
            self.checkpoint = None
